          btn.className = "px-3 sm:px-4 py-2 rounded-xl text-sm font-semibold transition-all text-slate-400 hover:text-white";
        }
      });
      lucide.createIcons();
    }

    // 렌더링 함수
    function renderAll() {
      // 1. 공지사항
      document.getElementById('notice-date-display').textContent = siteData.notice.date;
      document.getElementById('notice-content-display').textContent = siteData.notice.content;

      // 2. 메인 퀵 스케줄
      if (siteData.schedules.length > 0) {
        document.getElementById('main-quick-schedule').textContent = 
          `${siteData.schedules[0].day} - ${siteData.schedules[0].title}`;
      }

      // 3. 세계관
      const worldBox = document.getElementById('world-content-display');
      if (!siteData.world || siteData.world.trim() === "") {
        worldBox.innerHTML = `
          <i data-lucide="scroll" class="w-12 h-12 text-slate-500 mb-3"></i>
          <p class="text-slate-400 text-sm font-medium">아직 등록된 세계관 설정이 없습니다.</p>
          <p class="text-slate-500 text-xs mt-1">공지사항의 '항'을 눌러 관리자 모드에서 언제든지 내용을 작성해 보세요.</p>
        `;
      } else {
        worldBox.innerHTML = `
          <div class="w-full text-left text-slate-300 text-sm sm:text-base leading-relaxed whitespace-pre-line">
            ${siteData.world}
          </div>
        `;
      }

      // 4. 스케줄 리스트
      const scheduleBox = document.getElementById('schedule-content-display');
      scheduleBox.innerHTML = siteData.schedules.map(s => `
        <div class="flex flex-col sm:flex-row sm:items-center justify-between p-4 bg-[#161524] rounded-xl border border-brand-border gap-2">
          <div class="space-y-1">
            <span class="text-xs font-mono text-brand-periwinkle font-semibold">${s.day}</span>
            <h4 class="text-base font-bold text-white">${s.title}</h4>
          </div>
          <span class="self-start sm:self-center px-2.5 py-1 rounded-md tag-box text-xs font-bold">${s.tag}</span>
        </div>
      `).join('');

      // 5. 멤버 리스트
      const memberBox = document.getElementById('members-list-display');
      memberBox.innerHTML = siteData.members.map(m => `
        <div class="bg-[#161524] p-5 rounded-xl border border-brand-border space-y-3 relative overflow-hidden">
          <div class="flex items-center gap-3">
            <div class="w-10 h-10 rounded-lg bg-brand-tag border border-brand-periwinkle/30 flex items-center justify-center text-brand-periwinkle font-bold">
              ${m.name.substring(0, 1)}
            </div>
            <div>
              <h4 class="font-bold text-white text-base">${m.name}</h4>
              <p class="text-xs text-brand-periwinkle font-medium">${m.role}</p>
            </div>
          </div>
          <p class="text-xs text-slate-400 leading-relaxed">${m.desc}</p>
        </div>
      `).join('');

      lucide.createIcons();
    }

    // 비밀번호 모달 열기 ('항' 클릭)
    function triggerAdminModal() {
      document.getElementById('admin-pass-input').value = "";
      document.getElementById('auth-error-msg').classList.add('hidden');
      document.getElementById('auth-modal').classList.remove('hidden');
      setTimeout(() => document.getElementById('admin-pass-input').focus(), 50);
    }

    function closeAuthModal() {
      document.getElementById('auth-modal').classList.add('hidden');
    }

    // 비밀번호 검증
    function verifyAdminPass() {
      const entered = document.getElementById('admin-pass-input').value;
      if (entered === ADMIN_PASS) {
        closeAuthModal();
        openAdminDashboard();
      } else {
        document.getElementById('auth-error-msg').classList.remove('hidden');
      }
    }

    // 관리자 대시보드 열기
    function openAdminDashboard() {
      // 폼에 기존 데이터 채우기
      document.getElementById('edit-notice-date').value = siteData.notice.date;
      document.getElementById('edit-notice-text').value = siteData.notice.content;
      document.getElementById('edit-world-text').value = siteData.world || "";
      
      renderAdminScheduleEditor();
      renderAdminMemberEditor();

      setAdminTab('notice');
      document.getElementById('admin-dashboard-modal').classList.remove('hidden');
      lucide.createIcons();
    }

    function closeAdminDashboard() {
      document.getElementById('admin-dashboard-modal').classList.add('hidden');
    }

    // 관리자 모달 내부 탭 전환
    function setAdminTab(tab) {
      ['notice', 'world', 'schedule', 'members'].forEach(t => {
        const panel = document.getElementById(`admin-panel-${t}`);
        const btn = document.getElementById(`admin-tab-btn-${t}`);
        if (t === tab) {
          panel.classList.remove('hidden');
          btn.className = "py-3 px-4 font-semibold text-brand-periwinkle border-b-2 border-brand-periwinkle";
        } else {
          panel.classList.add('hidden');
          btn.className = "py-3 px-4 font-semibold text-slate-400 hover:text-white";
        }
      });
      lucide.createIcons();
    }

    // 스케줄 에디터 렌더링
